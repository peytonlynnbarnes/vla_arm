# =============================================================================
# TERM 0 — Kill everything from a prior session (always safe to run first)
# =============================================================================
# This is the only reliable way to recover from zombie sims: my repeated
# pkill/relaunch cycles earlier left parallel gz sim servers + ghost
# controller_manager entries on DDS, which caused "Controller already loaded,
# skipping load_controller" → FATAL on every subsequent launch.

set +e
pkill -9 -f "ros2 launch"        2>/dev/null || true
pkill -9 -f "ros2 run"           2>/dev/null || true
pkill -9 -f "gz sim"             2>/dev/null || true
pkill -9 -f "ruby.*gz_tools"     2>/dev/null || true
pkill -9 -f "parameter_bridge"   2>/dev/null || true
pkill -9 -f "robot_state_publisher" 2>/dev/null || true
pkill -9 -f "controller_manager" 2>/dev/null || true
pkill -9 -f "ros2cli.daemon"     2>/dev/null || true
pkill -9 -f "vla_action_client"  2>/dev/null || true
pkill -9 -f "action_to_ee"       2>/dev/null || true
pkill -9 -f "vla_ik"             2>/dev/null || true
sleep 4

# If anything survives (pkill -f won't match processes whose parent is PID 1
# once their launcher dies), kill by explicit PID from this list:
ps -ef | grep -E "gz sim|parameter_bridge|robot_state_publisher|ruby.*gz|vla_|python.*vla" | grep -v grep

# Then:
#   kill -9 <pid1> <pid2> ...
# For example (REPLACE pids):
#   kill -9 40271 40272 40273 40274 40275 40331 40332

# Optional: reset ros2 daemon so stale DDS entries can't confuse the next
# controller_manager. Only needed after a sim crash.
source /opt/ros/jazzy/setup.bash
ros2 daemon stop 2>&1 || true
sleep 2
ros2 daemon start 2>&1

# =============================================================================
# NOTES ON KNOWN GOTCHAS
# =============================================================================
#
# 1. The VLA trajectory targets in your last run walked around
#      (x ≈ 0.44–0.47, y ≈ -0.05, z ≈ 0.22)
#    while the blue ball sits at (0.22, +0.10, 0.11). OpenVLA doesn't
#    "know" where the blue ball is in this scene (SO-101 + colored spheres
#    are OOD for bridge_orig / Bridge V2 training data).
#    Mitigations:
#      - set translation_scale ≈ 0.01 in action_to_ee (shrink delta magnitude)
#      - seed a "home pose" that hovers the gripper near the ball before
#        VLA takes over, so VLA's small corrections can actually close the gap
#      - accept that without fine-tuning this pipeline demonstrates the
#        PLUMBING, not task competency.
#
# 2. If you change ros2_control controllers.yaml or the URDF's
#    <ros2_control> block, you MUST rebuild arm_description AND restart
#    Gazebo — gz_ros2_control reads these at plugin init only.
#
# 3. If you see "Controller already loaded" + "Failed to configure" and
#    `ros2 control list_controllers` shows "No controllers are currently
#    loaded!" (not both active), you almost certainly have a ghost
#    controller_manager from a prior sim — go back to TERM 0 and reset the
#    ros2 daemon.
#
# 4. Physics step in ball_world.sdf is 0.004 s (4 ms). Smaller values (e.g.
#    0.001) drop RTF proportionally. Don't touch unless you need finer
#    contact resolution.
